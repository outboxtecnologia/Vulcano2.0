export const RecebimentosView = ({ selectedEmpresa }) => {
  const [data, setData] = useState([]);
  const [loading, setLoading] = useState(false);
  const [empreendimentoFilter, setEmpreendimentoFilter] = useState('');
  const [dateFrom, setDateFrom] = useState('');
  const [dateTo, setDateTo] = useState('');
  const [activeTab, setActiveTab] = useState('RECEBER'); // 'RECEBER' or 'RECEBIDOS'
  const [baixaModal, setBaixaModal] = useState(null);
  const [showExportMenu, setShowExportMenu] = useState(false);
  
  // Pagination State
  const [currentPage, setCurrentPage] = useState(1);
  const ITEMS_PER_PAGE = 30;

  const [empreendimentosList, setEmpreendimentosList] = useState([]);

  useEffect(() => {
    if (!selectedEmpresa) return;
    fetch(`${API_BASE}/api/vulcano/empreendimentos?empresa_id=${selectedEmpresa}`)
      .then(res => res.json())
      .then(d => {
         const list = d || [];
         setEmpreendimentosList(list);
         if (list.length > 0) setEmpreendimentoFilter('Todos os Projetos');
      })
      .catch(console.error);
    setData([]);
  }, [selectedEmpresa]);

  const buscarRecebimentos = () => {
    if (!dateFrom || !dateTo) {
        alert("Selecione o Período Inicial e Final para buscar.");
        return;
    }
    const empId = empreendimentoFilter !== 'Todos os Projetos' 
       ? empreendimentosList.find(e => e.nome === empreendimentoFilter)?.id || ''
       : '';

    setLoading(true);
    fetch(`${API_BASE}/api/vulcano/recebimentos?empresa_id=${selectedEmpresa}&empreendimento_id=${empId}&data_ini=${dateFrom}&data_fim=${dateTo}`)
      .then(res => res.json())
      .then(d => {
        if (!Array.isArray(d)) {
           setData([]); setLoading(false); return;
        }
        setData(d);
        setCurrentPage(1);
        setLoading(false);
      })
      .catch(err => { console.error(err); setLoading(false); });
  };

  const dataAberto = data.filter(r => r.status_sistema === 'ABERTO' || r.total_pago <= 0);
  const dataConferencia = data.filter(r => r.status_sistema !== 'ABERTO' && r.total_pago > 0);
  
  const currentLevelData = activeTab === 'RECEBER' ? dataAberto : dataConferencia;

  useEffect(() => {
    setCurrentPage(1);
  }, [activeTab]);

  // KPIs Calculations
  const totalPendente = activeTab === 'RECEBER' 
    ? dataAberto.reduce((acc, curr) => acc + (curr.parcela || 0) + (curr.variacao || 0), 0)
    : dataConferencia.reduce((acc, curr) => acc + (curr.parcela || 0) + (curr.variacao || 0), 0);
  
  const inadimplencia = dataAberto.filter(r => {
     if (!r.data) return false;
     const parts = r.data.split('/');
     if (parts.length === 3) {
        const vencDate = new Date(`${parts[2]}-${parts[1]}-${parts[0]}T00:00:00`);
        const today = new Date(); today.setHours(0,0,0,0);
        return vencDate < today;
     }
     return false;
  }).reduce((acc, curr) => acc + (curr.parcela || 0), 0);
  
  const totalConfirmados = dataConferencia.reduce((acc, curr) => acc + (curr.total || 0), 0);

  const totalPages = Math.ceil(currentLevelData.length / ITEMS_PER_PAGE);
  const paginatedData = currentLevelData.slice((currentPage - 1) * ITEMS_PER_PAGE, currentPage * ITEMS_PER_PAGE);

  const handleDarBaixa = (r) => {
    if (!r.id) return alert("Parcela não possui ID vinculado.");
    setBaixaModal(r);
  };

  const submitBaixa = async (e) => {
    e.preventDefault();
    const fd = new FormData(e.target);
    try {
      setLoading(true);
      await fetch(`${API_BASE}/api/vulcano/recebimentos/baixa`, {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ 
           id_receber: baixaModal.id, 
           valor_pago: parseFloat(fd.get("valor_pago")) || 0,
           data_pagamento: fd.get("data_pagamento"),
           acrescimos: parseFloat(fd.get("acrescimos")) || 0,
           descontos: parseFloat(fd.get("descontos")) || 0,
           empresa_id: selectedEmpresa
        })
      });
      setBaixaModal(null);
      buscarRecebimentos();
    } catch (err) {
      alert("Erro ao dar baixa");
      setLoading(false);
    }
  };

  const handleExport = (type) => {
     const rows = currentLevelData.map(e => {
        return `${e.data}\t${e.total || 0}\t${e.parcela || 0}\t${e.variacao || 0}\t"${e.descricao_venda}"\t"${e.cliente}"\t"${e.obs || e.status_sistema}"`;
     });
     const header = "Vencimento\tPago\tParcela\tVariacao\tUnidade\tComprador\tOrigem";
     const content = header + "\n" + rows.join("\n");
     const encodedUri = encodeURI("data:text/plain;charset=utf-8," + content);
     const link = document.createElement("a");
     link.setAttribute("href", encodedUri);
     link.setAttribute("download", `recebimentos_${activeTab.toLowerCase()}.${type}`);
     document.body.appendChild(link);
     link.click();
     document.body.removeChild(link);
     setShowExportMenu(false);
  };

  return (
    <div className="flex flex-col h-full animate-in fade-in slide-in-from-bottom-4 duration-700 bg-[#0c0c0c] text-white">
      {/* HEADER STITCH */}
      <div className="flex justify-between items-center mb-6">
        <h2 className="text-3xl font-bold tracking-tight text-[#f5f5f5]">Recebimentos e Baixas</h2>
        <div className="relative">
          <button 
             onClick={() => setShowExportMenu(!showExportMenu)}
             className="bg-[#ff4d00] hover:bg-[#ff4d00]/90 text-black px-6 py-2.5 rounded text-xs tracking-wider uppercase font-black transition-colors flex items-center gap-2"
          >
             <Download size={16}/> Exportar Lista
          </button>
          {showExportMenu && (
             <div className="absolute right-0 mt-2 w-48 bg-[#1A262E] border border-white/10 rounded overflow-hidden shadow-xl z-[60]">
               <button onClick={() => handleExport('xlsx')} className="w-full text-left px-4 py-3 text-xs tracking-widest font-bold hover:bg-white/5 transition-colors border-b border-white/5 text-[#fff]">EXCEL (XLSX)</button>
               <button onClick={() => handleExport('txt')} className="w-full text-left px-4 py-3 text-xs tracking-widest font-bold hover:bg-white/5 transition-colors text-[#fff]">TEXTO / CSV</button>
             </div>
          )}
        </div>
      </div>

      {/* FILTER RIBBON */}
      <div className="flex gap-4 mb-8 bg-[#121212] p-4 rounded border border-white/5">
         <div className="flex-col max-w-xs w-full">
            <label className="text-[10px] font-bold uppercase tracking-widest text-[#777] mb-1.5 block">Empreendimento</label>
            <div className="relative">
               <select value={empreendimentoFilter} onChange={e => setEmpreendimentoFilter(e.target.value)} className="w-full bg-[#0a0a0a] text-white border border-white/10 rounded p-3 text-sm appearance-none cursor-pointer focus:outline-none focus:border-[#ff4d00]/50 font-mono">
                  <option value="Todos os Projetos">Todos os Projetos</option>
                  {empreendimentosList.map(e => <option key={e.id} value={e.nome}>{e.nome}</option>)}
               </select>
               <ChevronDown size={14} className="absolute right-3 top-1/2 -translate-y-1/2 text-[#777] pointer-events-none"/>
            </div>
         </div>
         <div className="flex-col w-48">
            <label className="text-[10px] font-bold uppercase tracking-widest text-[#777] mb-1.5 block">Data Início</label>
            <input type="date" value={dateFrom} onChange={e => setDateFrom(e.target.value)} className="w-full bg-[#0a0a0a] text-white border border-white/10 rounded p-3 text-sm focus:outline-none focus:border-[#ff4d00]/50 font-mono" />
         </div>
         <div className="flex-col w-48">
            <label className="text-[10px] font-bold uppercase tracking-widest text-[#777] mb-1.5 block">Data Fim</label>
            <input type="date" value={dateTo} onChange={e => setDateTo(e.target.value)} className="w-full bg-[#0a0a0a] text-white border border-white/10 rounded p-3 text-sm focus:outline-none focus:border-[#ff4d00]/50 font-mono" />
         </div>
         <div className="flex items-end">
            <button onClick={buscarRecebimentos} disabled={loading} className="bg-[#ff4d00] hover:bg-[#ff4d00]/90 text-black font-black text-xs px-8 py-3 rounded uppercase tracking-widest transition-colors flex items-center gap-2 h-[46px]">
               {loading ? <Loader2 size={16} className="animate-spin text-black" /> : <Filter size={16} />} APLICAR FILTROS
            </button>
         </div>
      </div>

      {/* TABS */}
      <div className="flex border-b border-white/10 mb-4 items-center pl-2">
         <button 
           onClick={() => setActiveTab('RECEBER')}
           className={`pb-3 px-4 font-bold tracking-wide transition-colors flex items-center gap-3 text-sm pt-2 ${activeTab === 'RECEBER' ? 'text-[#ff4d00] border-b-2 border-[#ff4d00]' : 'text-gray-500 hover:text-white'}`}
         >
            A Receber <span className={`text-[10px] rounded px-3 py-1 font-black ${activeTab === 'RECEBER' ? 'bg-[#ff4d00]/20 text-[#ff4d00]' : 'bg-white/5 text-gray-500'}`}>{dataAberto.length}</span>
         </button>
         <button 
           onClick={() => setActiveTab('RECEBIDOS')}
           className={`pb-3 px-6 font-bold tracking-wide transition-colors flex items-center gap-2 text-sm pt-2 ml-4 ${activeTab === 'RECEBIDOS' ? 'text-[#ff4d00] border-b-2 border-[#ff4d00]' : 'text-gray-500 hover:text-white'}`}
         >
            Recebidos
         </button>
      </div>

      {/* DATA GRID */}
      <div className="bg-[#111111] border border-white/5 rounded-sm flex flex-col flex-1 overflow-hidden relative min-h-[300px]">
         <div className="overflow-auto flex-1">
            <table className="w-full text-left text-xs border-collapse">
               <thead className="bg-[#181818] sticky top-0 z-10 border-b border-white/10">
                  <tr>
                    <th className="py-4 px-6 text-[10px] font-black text-[#777] uppercase tracking-[0.2em]">Vencimento</th>
                    <th className="py-4 px-6 text-[10px] font-black text-[#777] uppercase tracking-[0.2em] text-right">Pago (R$)</th>
                    <th className="py-4 px-6 text-[10px] font-black text-[#777] uppercase tracking-[0.2em] text-right">Parcela (R$)</th>
                    <th className="py-4 px-6 text-[10px] font-black text-[#777] uppercase tracking-[0.2em] text-right">Variação (R$)</th>
                    <th className="py-4 px-6 text-[10px] font-black text-[#777] uppercase tracking-[0.2em]">Unidade</th>
                    <th className="py-4 px-6 text-[10px] font-black text-[#777] uppercase tracking-[0.2em]">Comprador</th>
                    <th className="py-4 px-6 text-[10px] font-black text-[#777] uppercase tracking-[0.2em]">Origem</th>
                    <th className="py-4 px-6 text-[10px] font-black text-[#777] uppercase tracking-[0.2em] text-center">Ações</th>
                  </tr>
               </thead>
               <tbody className="divide-y divide-white/5">
                  {paginatedData.map((r, idx) => {
                     const varNum = r.variacao || 0;
                     const isNegative = varNum < 0;
                     let originText = r.obs ? r.obs.substring(0,20) : (r.status_sistema || 'Boleto');
                     if (originText.toUpperCase() === 'ABERTO') originText = "Boleto";
                     return (
                        <tr key={idx} className="hover:bg-white/[0.02] transition-colors">
                           <td className="py-4 px-6 text-[#999] font-mono tracking-wider">{r.data}</td>
                           <td className="py-4 px-6 text-right font-mono text-[#999] tracking-wider">{formatCurrency(r.total || 0).replace('R$ ', '')}</td>
                           <td className="py-4 px-6 text-right font-mono text-white tracking-widest">{formatCurrency(r.parcela || 0).replace('R$ ', '')}</td>
                           <td className={`py-4 px-6 text-right font-mono tracking-widest ${varNum === 0 ? 'text-[#555]' : isNegative ? 'text-[#ff3b30]' : 'text-[#60A5FA]'}`}>
                              {varNum > 0 && '+'}{formatCurrency(varNum).replace('R$ ', '')}
                           </td>
                           <td className="py-4 px-6 text-[#999] font-mono text-xs uppercase max-w-[140px] truncate">{r.descricao_venda}</td>
                           <td className="py-4 px-6 text-white max-w-[200px] truncate">{r.cliente}</td>
                           <td className="py-4 px-6">
                              <span className="text-[8px] uppercase tracking-[0.2em] font-bold bg-white/5 text-[#888] rounded px-2 py-1">
                                 {originText}
                              </span>
                           </td>
                           <td className="py-4 px-6 text-center flex justify-center">
                              {activeTab === 'RECEBER' ? (
                                <button onClick={() => handleDarBaixa(r)} className="flex items-center gap-2 bg-[#1A262E] hover:bg-white hover:text-black text-[#888] border border-white/5 rounded px-4 py-2 text-[8px] font-black tracking-[0.2em] transition-all">
                                   <CheckCircle size={12} className="text-[#60A5FA]"/> DAR BAIXA
                                </button>
                              ) : (
                                <span className="text-[9px] text-[#555] font-black uppercase tracking-widest bg-white/5 px-4 py-1.5 rounded"><CheckCircle size={10} className="inline mr-1 text-[#34c759]"/> Concluído</span>
                              )}
                           </td>
                        </tr>
                     );
                  })}
                  {paginatedData.length === 0 && !loading && (
                     <tr><td colSpan="8" className="py-16 text-center text-[#555] uppercase tracking-widest text-[10px] bg-white/[0.01]">Utilize os filtros acima para carregar a carteira.</td></tr>
                  )}
               </tbody>
            </table>
         </div>
         <div className="p-4 border-t border-white/5 flex justify-between items-center bg-[#181818]">
             <span className="text-[9px] text-[#555] uppercase font-black tracking-[0.2em]">
                EXIBINDO {paginatedData.length} DE {currentLevelData.length} REGISTROS PENDENTES
             </span>
             <div className="flex gap-1.5 items-center">
                <button onClick={() => setCurrentPage(prev => Math.max(1, prev - 1))} disabled={currentPage === 1} className="bg-[#222] p-1.5 px-3 text-[#777] rounded hover:text-white disabled:opacity-30 disabled:hover:text-[#777] transition-colors"><ChevronLeft size={16}/></button>
                <div className="flex items-center bg-[#ff4d00] text-black font-black h-8 w-8 justify-center rounded text-xs mx-2 shadow-[0_0_10px_rgba(255,77,0,0.3)]">{currentPage}</div>
                <button onClick={() => setCurrentPage(prev => Math.min(totalPages, prev + 1))} disabled={currentPage >= totalPages && totalPages > 0} className="bg-[#222] p-1.5 px-3 text-[#777] rounded hover:text-white disabled:opacity-30 disabled:hover:text-[#777] transition-colors"><ChevronRight size={16}/></button>
             </div>
         </div>
      </div>

      {/* FOOTER KPIs */}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-6 mt-8 shrink-0">
         <div className="border border-white/5 bg-[#121212] rounded p-6 relative">
            <h4 className="text-[10px] text-[#888] font-bold uppercase tracking-[0.2em] mb-2">{activeTab === 'RECEBER' ? 'TOTAL A RECEBER (MÊS)' : 'TOTAL HISTÓRICO'}</h4>
            <div className="text-3xl font-bold tracking-tight text-white mb-2">{formatCurrency(totalPendente)}</div>
            <p className="text-[10px] font-bold text-[#60A5FA]"><TrendingUp size={12} className="inline mr-1"/> Indicador em Tempo Real</p>
         </div>
         <div className="border border-white/5 bg-[#121212] rounded p-6 relative border-l border-l-[#ff3b30]">
            <h4 className="text-[10px] text-[#888] font-bold uppercase tracking-[0.2em] mb-2">PROJEÇÃO DE INADIMPLÊNCIA</h4>
            <div className="text-3xl font-bold tracking-tight text-white mb-2">{formatCurrency(inadimplencia)}</div>
            <p className="text-[10px] font-bold text-[#ff3b30] bg-[#ff3b30]/10 inline-block px-2 py-0.5 rounded"><AlertCircle size={10} className="inline mr-1"/> Atenção: {dataAberto.filter(r => {
                 if (!r.data) return false;
                 const parts = r.data.split('/');
                 if (parts.length === 3) {
                    const vencDate = new Date(`${parts[2]}-${parts[1]}-${parts[0]}T00:00:00`);
                    const today = new Date(); today.setHours(0,0,0,0);
                    return vencDate < today;
                 }
                 return false;
              }).length} unidades com > atraso</p>
         </div>
         <div className="border border-white/5 bg-[#121212] rounded p-6 relative flex justify-between items-center pr-10">
            <div>
               <h4 className="text-[10px] text-[#888] font-bold uppercase tracking-[0.2em] mb-2">RECEBIMENTOS CONFIRMADOS</h4>
               <div className="text-3xl font-bold tracking-tight text-white mb-2">{formatCurrency(totalConfirmados)}</div>
               <p className="text-[10px] text-[#555] uppercase font-bold tracking-widest">Valores conciliados via EDI bancário</p>
            </div>
            <div className="bg-[#ff4d00] text-black w-14 h-14 flex items-center justify-center rounded-sm absolute right-6 shadow-[0_0_20px_rgba(255,77,0,0.2)]">
               <Zap size={24} className="fill-current text-black"/>
            </div>
         </div>
      </div>

      {/* MODAL BAIXA EFETIVA */}
      {baixaModal && (
         <div className="fixed inset-0 z-[500] flex items-center justify-center bg-black/95 p-6 animate-in fade-in backdrop-blur-sm">
           <form onSubmit={submitBaixa} className="bg-[#121212] w-full max-w-[450px] flex flex-col border border-white/10 p-8 rounded-sm relative shadow-[0_0_40px_rgba(255,77,0,0.1)]">
             <div className="flex justify-between items-start mb-8">
               <div>
                  <h3 className="text-xl font-bold uppercase tracking-widest text-[#f5f5f5] flex items-center gap-2">
                    Efetivar Baixa Fiscal
                  </h3>
                  <p className="text-[10px] text-[#ff4d00] uppercase tracking-[0.2em] mt-2 font-bold">Parcela: R$ {baixaModal.parcela} • Venc: {baixaModal.data}</p>
               </div>
               <button type="button" onClick={() => setBaixaModal(null)} className="text-[#555] hover:text-white p-2 transition-colors">
                 <X size={20}/>
               </button>
             </div>
             
             <div className="space-y-5 mb-8 relative">
               <div className="bg-black p-4 rounded-sm border border-white/5">
                 <span className="text-[10px] font-bold block mb-1 uppercase text-[#666] tracking-widest">{baixaModal.cliente}</span>
                 <span className="text-xs text-white uppercase tracking-[0.2em] font-black">{baixaModal.descricao_venda}</span>
               </div>
               
               <div>
                 <label className="text-[10px] font-bold uppercase tracking-[0.2em] text-[#777] block mb-2">Data Efetiva do Pagamento</label>
                 <input type="date" name="data_pagamento" required defaultValue={new Date().toISOString().slice(0,10)} className="w-full bg-[#0a0a0a] text-white border border-white/10 rounded-sm p-3.5 font-mono text-sm focus:outline-none focus:border-[#ff4d00]/50" />
               </div>
               
               <div className="flex gap-4">
                 <div className="flex-1">
                   <label className="text-[10px] font-bold uppercase tracking-[0.2em] text-[#ff3b30] block mb-2">Descontos (-) R$</label>
                   <input type="number" step="0.01" name="descontos" defaultValue="0" min="0" className="w-full bg-[#0a0a0a] text-white border border-white/10 rounded-sm p-3.5 text-right focus:outline-none focus:border-[#ff4d00]/50" />
                 </div>
                 <div className="flex-1">
                   <label className="text-[10px] font-bold uppercase tracking-[0.2em] text-[#60A5FA] block mb-2">Acréscimos (+) R$</label>
                   <input type="number" step="0.01" name="acrescimos" defaultValue="0" min="0" className="w-full bg-[#0a0a0a] text-white border border-white/10 rounded-sm p-3.5 text-right focus:outline-none focus:border-[#ff4d00]/50" />
                 </div>
               </div>
               
               <div className="pt-2">
                 <label className="text-[10px] font-bold uppercase tracking-[0.2em] text-[#ff4d00] block mb-2">Valor Total Depositado (Liquidez)</label>
                 <input type="number" step="0.01" name="valor_pago" required defaultValue={baixaModal.parcela} className="w-full bg-[#ff4d00]/5 text-[#ff4d00] border border-[#ff4d00]/50 rounded-sm p-4 font-mono text-2xl font-bold text-right focus:outline-none focus:border-[#ff4d00]" />
               </div>
             </div>
             
             <button type="submit" disabled={loading} className="w-full bg-[#ff4d00] text-black font-black uppercase tracking-[0.2em] text-xs py-5 rounded-sm hover:bg-white transition-colors flex items-center justify-center gap-2">
               {loading ? <Loader2 size={16} className="animate-spin" /> : <CheckCircle2 size={16}/>} Confirmar Recebimento
             </button>
           </form>
         </div>
      )}
    </div>
  );
};
